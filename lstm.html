  // FUNGSI PEMROSESAN VEKTOR ANGIN, KECEPATAN & PARSING FITUR
  function processRawData(rawData) {
    return rawData.map(item => {
      const w = Number(item.water_level ?? item.tinggi_muka_air ?? item.tinggi_air ?? 0) || 0;
      const t = Number(item.temp ?? item.suhu ?? 0) || 0;
      const h = Number(item.humidity ?? item.kelembapan ?? 0) || 0;
      const p = Number(item.pressure ?? item.tekanan ?? 0) || 0;
      const deg = Number(item.wind_dir ?? item.arah_angin ?? item.deg ?? 0) || 0;
      const s = Number(item.wind_speed ?? item.kecepatan_angin ?? item.speed ?? 0) || 0;

      // Transformasi Derajat Angin ke Vektor Sinus & Kosinus
      const rad = (deg * Math.PI) / 180;
      return {
        water_level: w,
        temp: t,
        humidity: h,
        pressure: p,
        wind_speed: s,
        wind_u: Math.sin(rad),
        wind_v: Math.cos(rad)
      };
    });
  }

  // FUNGSI PERHITUNGAN METRIK AKURASI (RMSE, MAE, R, R²)
  function calculateMetrics(yTrue, yPred) {
    const n = yTrue.length;
    if (n === 0) return { rmse: 0, mae: 0, r: 0, r2: 0 };

    let sumErr2 = 0;
    let sumAbsErr = 0;
    let sumTrue = 0;
    let sumPred = 0;

    for (let i = 0; i < n; i++) {
      const err = yTrue[i] - yPred[i];
      sumErr2 += err * err;
      sumAbsErr += Math.abs(err);
      sumTrue += yTrue[i];
      sumPred += yPred[i];
    }

    const meanTrue = sumTrue / n;
    const meanPred = sumPred / n;

    let numR = 0;
    let denTrueR = 0;
    let denPredR = 0;
    let ssTot = 0;

    for (let i = 0; i < n; i++) {
      const diffTrue = yTrue[i] - meanTrue;
      const diffPred = yPred[i] - meanPred;

      numR += diffTrue * diffPred;
      denTrueR += diffTrue * diffTrue;
      denPredR += diffPred * diffPred;
      ssTot += diffTrue * diffTrue;
    }

    const rmse = Math.sqrt(sumErr2 / n);
    const mae = sumAbsErr / n;

    const denR = Math.sqrt(denTrueR) * Math.sqrt(denPredR);
    const r = denR === 0 ? 0 : numR / denR;

    const r2 = ssTot === 0 ? 0 : 1 - (sumErr2 / ssTot);

    return { rmse, mae, r, r2 };
  }

      if (!dataVal) {
        snapshot = await db.ref('cuaca/history').once('value');
        dataVal = snapshot.val();
      }

      if (!dataVal) throw new Error("Data histori tidak ditemukan di database.");

      let rawList = Array.isArray(dataVal) ? dataVal : Object.values(dataVal);
      let cleanDataset = rawList.filter(d => d && typeof d === 'object');

      if (cleanDataset.length < 10) throw new Error("Dataset terlalu sedikit (butuh minimal 10 data histori).");

      statusText.innerText = `Memproses ${cleanDataset.length} sampel data...`;
      const processedData = processRawData(cleanDataset);

      // Gunakan seluruh data yang tersedia (misalnya 79 data)
      const dataset = processedData.slice(-79);

      const features = dataset.map(d => [
        d.water_level, d.temp, d.humidity, d.pressure, d.wind_speed, d.wind_u, d.wind_v
      ]);

      const numFeatures = 7; // 7 Fitur utama termasuk kecepatan_angin dan vektor angin
      const timeSteps = 5;       // Lookback 5 langkah
      const forecastSteps = 12;  // Prediksi 12 langkah ke depan

      // Normalisasi Min-Max dengan Proteksi Division by Zero
      const mins = Array(numFeatures).fill(Infinity);
      const maxs = Array(numFeatures).fill(-Infinity);

      for (let i = 0; i < features.length; i++) {
        for (let j = 0; j < numFeatures; j++) {
          if (features[i][j] < mins[j]) mins[j] = features[i][j];
          if (features[i][j] > maxs[j]) maxs[j] = features[i][j];
        }
      }

      const scaledFeatures = features.map(row =>
        row.map((val, col) => {
          const range = maxs[col] - mins[col];
          return range === 0 ? 0 : (val - mins[col]) / range;
        })
      );

      // Form Dataset Training Sekuensial
      const X_train = [], Y_train = [];
      for (let i = 0; i <= scaledFeatures.length - timeSteps - 1; i++) {
        X_train.push(scaledFeatures.slice(i, i + timeSteps));
        Y_train.push(scaledFeatures[i + timeSteps]);
      }

      const ys = tf.tensor2d(Y_train);

      const model = tf.sequential();
      model.add(tf.layers.lstm({
        units: 16,
        inputShape: [timeSteps, numFeatures],
        returnSequences: false
      }));

    }

  function updateUI(historicalData, predictions) {
    if (predictions.length > 0) {
      document.getElementById('predWater').innerText = predictions[0].water_level.toFixed(2) + " m";
      document.getElementById('predTemp').innerText = predictions[0].temp.toFixed(1) + " °C";
      document.getElementById('predHum').innerText = predictions[0].humidity.toFixed(1) + " %";
      document.getElementById('predWindDir').innerText = predictions[0].wind_dir.toFixed(0) + "°";
    }

    const histWater = historicalData.map(d => d.water_level);
    const histLabels = historicalData.map((_, idx) => `D-${historicalData.length - idx}`);
    const predLabels = predictions.map((_, idx) => `P+${idx + 1}`);

    waterChart.data.labels = [...histLabels, ...predLabels];
    waterChart.data.datasets[0].data = [...histWater, ...Array(predictions.length).fill(null)];
    waterChart.data.datasets[1].data = [
      ...Array(histWater.length - 1).fill(null),
      histWater[histWater.length - 1],
      ...predictions.map(p => p.water_level)
    ];
    waterChart.update();

    humChart.data.labels = predLabels;
    humChart.data.datasets[0].data = predictions.map(p => p.humidity);
    humChart.data.datasets[1].data = predictions.map(p => p.temp);
    humChart.update();

    windChart.data.labels = predLabels;
    windChart.data.datasets[0].data = predictions.map(p => p.wind_dir);
    windChart.data.datasets[1].data = predictions.map(p => p.pressure);
    windChart.update();

    const tableBody = document.getElementById('predictionTableBody');
    tableBody.innerHTML = '';

    predictions.forEach((p, idx) => {
      tableBody.innerHTML += `
        
          Langkah +${idx + 1}
          ${p.water_level.toFixed(2)}
          ${p.temp.toFixed(1)}
          ${p.humidity.toFixed(1)}
          ${p.pressure.toFixed(1)}
          ${p.wind_dir.toFixed(0)}°
        
      `;
    });
  }
